cmd_030_caesar_cipher() {

read -r -p "$(echo -e "${C_YELLOW}Décalage (nombre) : ${C_RESET}")" shift
shift=${shift:-3}
local result=""
local i ch
for ((i=0; i<${#text}; i++)); do
ch="${text:$i:1}"
if [[ "$ch" =~ [A-Z] ]]; then
result+=$(printf "\\$(printf '%03o' $(( ( $(printf '%d' "'$ch") - 65 + shift) % 26 + 65 )))")
elif [[ "$ch" =~ [a-z] ]]; then
result+=$(printf "\\$(printf '%03o' $(( ( $(printf '%d' "'$ch") - 97 + shift) % 26 + 97 )))")
else
result+="$ch"
fi
done
echo -e "\n${C_GREEN}${result}${C_RESET}"
}

cmd_033_prime_check() {
read -r -p "$(echo -e "${C_YELLOW}Nombre : ${C_RESET}")" n
if ! [[ "$n" =~ ^[0-9]+$ ]]; then
err "Nombre invalide"
return
fi
if [ "$n" -lt 2 ]; then
echo -e "${C_RED}${n} n'est pas premier${C_RESET}"
return
fi
local i=2
local sqrt
sqrt=$(echo "sqrt($n)" | bc 2>/dev/null)
while [ "$i" -le "$sqrt" ]; do
if [ $((n % i)) -eq 0 ]; then
echo -e "${C_RED}${n} n'est pas premier (divisible par ${i})${C_RESET}"
return
fi
i=$((i + 1))
done
echo -e "${C_GREEN}${n} est PREMIER${C_RESET}"
}

cmd_036_prime_factors() {
read -r -p "$(echo -e "${C_YELLOW}Nombre : ${C_RESET}")" n
if ! [[ "$n" =~ ^[0-9]+$ ]] || [ "$n" -lt 2 ]; then
err "Nombre invalide"
return
fi
local original=$n
local factors=()
local d=2
while [ $((d * d)) -le "$n" ]; do
while [ $((n % d)) -eq 0 ]; do
factors+=("$d")
n=$((n / d))
done
d=$((d + 1))
done
if [ "$n" -gt 1 ]; then
factors+=("$n")
fi
local IFS=" × "
echo -e "\n${C_GREEN}${original} = ${factors[*]}${C_RESET}"
}

cmd_037_gcd_lcm() {
read -r -p "$(echo -e "${C_YELLOW}a : ${C_RESET}")" a
read -r -p "$(echo -e "${C_YELLOW}b : ${C_RESET}")" b
if ! [[ "$a" =~ ^[0-9]+$ ]] || ! [[ "$b" =~ ^[0-9]+$ ]]; then
err "Nombres invalides"
return
fi
# PGCD via Euclide
local x=$a y=$b tmp
while [ "$y" -ne 0 ]; do
tmp=$y
y=$((x % y))
x=$tmp
done
local gcd=$x
local lcm=$((a * b / gcd))
echo -e "\n${C_GREEN}PGCD(${a},${b}) = ${gcd}${C_RESET}"
echo -e "${C_GREEN}PPCM(${a},${b}) = ${lcm}${C_RESET}"
}

local range=$((high - low + 1))

}

t1=$(date +%s%N)

echo " seq + awk : ${dt3}ms"
echo ""
local fastest="for"

# ═══════════════════════════════════════════════════════════════════════════════
# MODE INTERACTIF
# ═══════════════════════════════════════════════════════════════════════════════

interactive_mode() {
while true; do
show_menu
printf "${C_YELLOW}➜ ${C_RESET}"
read -r choice
choice="${choice^^}"

if [ -z "$choice" ]; then
continue
fi

case "$choice" in
Q|QUIT|EXIT)
echo -e "\n${C_MAGENTA}👋 À bientôt !${C_RESET}\n"
break
;;
H)
show_help
continue
;;
L)
list_all_commands
continue
;;
esac

# Numéro
if [[ "$choice" =~ ^[0-9]+$ ]]; then
local n="$choice"

# Catégorie
if [ "$n" -ge 1 ] && [ "$n" -le 9 ]; then
local name="${CATEGORIES[$n]%|*}"
local range="${CATEGORIES[$n]#*|}"
local start="${range% *}"
local end="${range#* }"

clear_screen
echo -e "${C_CYAN}Catégorie : ${name}${C_RESET}\n"
local c
for ((c=start; c<=end; c++)); do
local entry="${COMMANDS[$c]:-}"
local cname="${entry%|*}"
printf " ${C_GREEN}[%03d]${C_RESET} %s\n" "$c" "$cname"
done
echo ""
read -r -p "$(echo -e "${C_YELLOW}Numéro de commande (ou Entrée pour retour) : ${C_RESET}")" sub
if [[ "$sub" =~ ^[0-9]+$ ]] && [ -n "${COMMANDS[$sub]:-}" ]; then
run_command "$sub"
fi
continue
fi

# Commande directe
if [ -n "${COMMANDS[$n]:-}" ]; then
run_command "$n"
continue
fi
fi

err "Choix invalide : $choice"
sleep 1
done
}

cli_mode() {
local arg="$1"

case "$arg" in
-h|--help)
show_help
return
;;
-l|--list)
list_all_commands
return
;;
-v|--version)
echo "${PROJECT_NAME} v${VERSION}"
echo "Auteur : ${AUTHOR}"
return
;;
esac

if [[ "$arg" == "--cat" ]]; then
local n="$2"
if [[ "$n" =~ ^[0-9]+$ ]] && [ "$n" -ge 1 ] && [ "$n" -le 9 ]; then
local name="${CATEGORIES[$n]%|*}"
local range="${CATEGORIES[$n]#*|}"
local start="${range% *}"
local end="${range#* }"
local c
for ((c=start; c<=end; c++)); do
local entry="${COMMANDS[$c]:-}"
local cname="${entry%|*}"
local fn="${entry#*|}"
echo ""
printf "${C_CYAN}${C_BOLD}[#%03d] %s${C_RESET}\n\n" "$c" "$cname"
if declare -F "$fn" >/dev/null 2>&1; then
"$fn"
fi
done
fi
return
fi

# Numéro de commande
if [[ "$arg" =~ ^[0-9]+$ ]] && [ -n "${COMMANDS[$arg]:-}" ]; then
local entry="${COMMANDS[$arg]}"
local name="${entry%|*}"
local fn="${entry#*|}"
clear_screen
printf "${C_CYAN}${C_BOLD}[#%03d] %s${C_RESET}\n\n" "$arg" "$name"
printf "${C_DIM}%s${C_RESET}\n\n" "$(hr '─' 78)"
if declare -F "$fn" >/dev/null 2>&1; then
"$fn"
fi
return
fi

err "Argument inconnu : $arg"
echo "Utilise --help pour l'aide"
}
